---
title: Get Payout
api: GET /api/payouts/{id}
description: Get a payout by ID
---

<Info>**Scope** `payouts:read`</Info>

## Headers

<ParamField header="Authorization" type="string" required>
  API key.
</ParamField>

## Path parameters

<ParamField path="id" type="string" required>
  The payout's `transactionId` returned from [`POST /rfq/execute`](/api-reference/payouts/execute).
</ParamField>

## Response

<ResponseField name="id" type="string">
  Payout ID.
</ResponseField>

<ResponseField name="transaction_type" type="string">
  `payout` or `swap`.
</ResponseField>

<ResponseField name="status" type="string">
  `pending`, `processing`, `completed`, `failed`, or `cancelled`. See [Core Concepts](/concepts#transaction-status) for the lifecycle.
</ResponseField>

<ResponseField name="source_amount" type="number">
  Amount debited.
</ResponseField>

<ResponseField name="source_currency" type="string">
  ISO 4217 code.
</ResponseField>

<ResponseField name="payout_amount" type="number">
  Amount credited to the recipient.
</ResponseField>

<ResponseField name="payout_currency" type="string">
  ISO 4217 code.
</ResponseField>

<ResponseField name="recipient_id" type="string">
  Recipient ID.
</ResponseField>

<ResponseField name="payment_method" type="string">
  Funding method.
</ResponseField>

<ResponseField name="payment_purpose" type="string">
  Stated purpose.
</ResponseField>

<ResponseField name="reference" type="string">
  Free-text reference you set at execution time.
</ResponseField>

<ResponseField name="provider_metadata" type="object">
  Routing detail for the executed payout.

  <Expandable title="provider_metadata">
    <ResponseField name="version" type="integer">
      Schema version.
    </ResponseField>
    <ResponseField name="onramp" type="object">
      Onramp leg detail. Present on fiat-to-stablecoin and fiat-to-fiat payouts.
    </ResponseField>
    <ResponseField name="offramp" type="object">
      Offramp leg detail. Present on stablecoin-to-fiat and fiat-to-fiat payouts.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="created_at" type="string">
  ISO 8601 timestamp.
</ResponseField>

<ResponseField name="completed_at" type="string">
  ISO 8601 timestamp. Present when `status` is `completed`.
</ResponseField>
